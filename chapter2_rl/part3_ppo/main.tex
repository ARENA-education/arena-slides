\section{Where we left off}

\begin{frame}
    \frametitle{Recap: VPG with importance sampling}
    Batch of full episodes collected from $\pi_{\bth_\old}$, then do gradient ascent on
    \begin{block}{VPG joy (with importance sampling)}
        $$
        J(\bth) = \hat{\Ex}_t \Big[ \red{r_t(\bth)} \, \blue{G_t} \Big],
        \qquad
        \red{r_t(\bth)} = \frac{\pi_\bth(a_t|s_t)}{\pi_{\bth_\old}(a_t|s_t)}
        $$
    \end{block}
    where $\hat{\Ex}_t$ averages over every timestep in the batch.
    %From now on the discount is back: $G_t = r_{t+1} + \gamma r_{t+2} + \gamma^2 r_{t+3} + \ldots$

    \pause
    \vspace{0.3cm}
    PPO keeps this shape, and fixes three problems:
    \begin{enumerate}
        \item \blue{$G_t$ is noisy.} It credits $a_t$ with every reward that followed,
        whether $a_t$ caused it or not. \hfill $\to$ \textbf{Ask a critic}
        \item \blue{$G_t$ needs full episode.} We want to learn from short, \\
        fixed-length rollouts. \hfill $\to$ \textbf{Use GAE}
        \item \red{$\pi_\bth$ can drift} far from $\pi_{\bth_\old}$ if batches
        reused. \hfill $\to$ \textbf{clamp $r_t(\bth)$}
    \end{enumerate}
    \pause
    Add an \textbf{entropy bonus} to keep exploring, and that's PPO.
\end{frame}

\section{Advantage and the critic}

\begin{frame}
    \frametitle{Advantages}
    \begin{itemize}
        \item In CartPole, every step of a long episode has a big $G_t$.
        So VPG pushes up \textit{every} action in it, good or bad.
        \pause
        \item $G_t$ = ``how good was the situation I was in?''
        \item Want: ``was the $a_t$ taken better/worse than what $\pi(\cdot \mid s_t)$ would \textit{usually} do?''
    \end{itemize}
    \begin{block}{Advantage}
        $$
        A_\pi(s_t, a_t) =
        \underbrace{Q_\pi(s_t,a_t)}_{\text{do } a_t \text{, then follow } \pi}
        - \underbrace{V_\pi(s_t)}_{\text{just follow } \pi}
        $$
    \end{block}
    \pause
    \begin{itemize}
        \item $A > 0$ : $a_t$ was better $\to$ push $\pi(a_t \mid s_t)$ up
        \item $A < 0$: $a_t$ was worse $\to$ pull $\pi(a_t \mid s_t)$ down
        \item $G_t$ is a sample of $Q_\pi(s_t,a_t)$, so $G_t - V_\pi(s_t)$
        is a sample of $A_\pi(s_t,a_t)$.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Subtracting a baseline is free}
    Why may we swap $G_t$ for $G_t - V_\pi(s_t)$?
    The extra term has zero expected gradient:
    \begin{align*}
        \onslide<2->{\Ex_{a \sim \pi_\bth}\Big[\nabla_\bth \log \pi_\bth(a|s) \, b(s)\Big]}
        \onslide<3->{&= \sum_a \pi_\bth(a|s) \frac{\nabla_\bth \pi_\bth(a|s)}{\pi_\bth(a|s)} b(s)} \\
        \onslide<4->{&= b(s) \sum_a \pi_\bth(a|s) \frac{\nabla_\bth \pi_\bth(a|s)}{\pi_\bth(a|s)}} \\
        \onslide<5->{&= b(s)\, \nabla_\bth \sum_a \pi_\bth(a|s)} \\
        \onslide<6->{&= b(s)\, \nabla_\bth 1 = 0}
    \end{align*}
    \onslide<6->{for any \textbf{baseline} $b(s)$ that depends only on the state.}

    \vspace{0.3cm}
    \onslide<7->{
    \begin{itemize}
        \item Mean gradient is the same, hopefully variance is lower.
        \item A good choice is $b(s) = V_\pi(s)$. But we don't know $V_\pi$\ldots
    \end{itemize}}
\end{frame}

\begin{frame}
    \frametitle{The critic}
    \ldots so learn it: add a second \textbf{head}, the \textbf{critic} $V_\pth(s)$, to the policy network.
    \begin{center}
    \begin{tikzpicture}[

    \end{tikzpicture}
    \end{center}
    \pause
    Train it by regression onto what actually happened:
    $$
    L^{VF}(\pth) = \hat{\Ex}_t\Big[\big(V_\pth(s_t) - \text{target}_t\big)^2\Big]
    $$
    with $\text{target}_t = G_t$ for now.
    \pause
    \begin{itemize}
        \item \red{Actor} $\pi_\bth$: chooses actions. Trained by policy gradient.
        \item \blue{Critic} $V_\pth$: judges states. Used to compute advantages.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Aside: $n$-step update}
    Complete return $G_t \equiv G_{t:T} = r_{t+1} + \gamma r_{t+2} + \ldots + \gamma^{T-t-1} r_T$.
    Can write approximations using $n$-step returns:
    $$
    G_{t:t+n} = r_{t+1} + \gamma r_{t+2} + \ldots + \gamma^{n-1} r_{t+n}  + \gamma^{n} V_\pi(s_{t+n})
    $$
    \pause
    In TD(0), we use the single-step return:
    \begin{align*}
        \delta_t &=  r_{t+1} + \gamma V_\pi(s_{t+1}) - V_\pi(s_t) \\
      &= G_{t:t+1}  - V_\pi(s_t)
    \end{align*}
    \pause
    Can be shown that the $n$-step returns $G_{t:t+n}$ are all valid targets for TD updates.
    \begin{itemize}
        \item Smaller $n$: Lower variance, higher bias
        \item Larger $n$: Higher variance, lower bias
        \item ...is there a way to get the benefits of both?
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{$n$-step: some dice, then the critic}
    $n$-step advantage: $n$ \red{real rewards}, then the \blue{critic's guess} for the rest.
    Everywhere $V_\pi$ appeared, use the critic as it was during the rollout, $\Vold$:
    $$
    A^{(n)}_t = G_{t:t+n} - \Vold(s_t) = \red{r_{t+1} + \ldots + \gamma^{n-1} r_{t+n}}
    + \blue{\gamma^n \Vold(s_{t+n})} - \Vold(s_t)
    $$
    \pause
    This is just a sum of TD errors. For $n = 2$:
    \begin{align*}
        \delta_t + \gamma \delta_{t+1}
        &= r_{t+1} + \cancel{\gamma \Vold(s_{t+1})} - \Vold(s_t) \\
        &\quad + \gamma r_{t+2} + \gamma^2 \Vold(s_{t+2}) - \cancel{\gamma \Vold(s_{t+1})} \\
        &= A^{(2)}_t
    \end{align*}
    \pause
    In general the sum telescopes:
    $$
    A^{(n)}_t = \delta_t + \gamma\delta_{t+1} + \ldots + \gamma^{n-1}\delta_{t+n-1}
    $$
    \pause
    \textbf{Intuition:} $\delta_k$ is the \textbf{surprise} on step $k$:
    how much better that step went than the critic expected.
    The \textbf{advantage} $A_\pi(s_t,a_t)$ of taking action $a_t$ is (in expectation) the
    discounted sum of the surprises that followed.
\end{frame}

\begin{frame}
    \frametitle{GAE: fade out the later surprises}
    Surprises far in the future are probably not caused by $a_t$, and they add noise.
    So fade them out:
    \begin{block}{Generalised Advantage Estimate (GAE)}
        \vspace{-0.4cm}
        \begin{align*}
            A_t &= (1-\lambda)\big(A^{(1)}_t + \lambda\,A^{(2)}_t + \lambda^2\,A^{(3)}_t + \ldots\big) \\
            &= \delta_t + (\gamma\lambda)\,\delta_{t+1} + (\gamma\lambda)^2\,\delta_{t+2} + \ldots
        \end{align*}
    \end{block}
    \pause
    \begin{columns}
    \begin{column}{0.5\linewidth}
    \begin{itemize}
        \item $\lambda = 0$: $A_t = \delta_t$. \\ Pure TD: trust the critic.
        \item $\lambda = 1$: $A_t = G_t - \Vold(s_t)$. \\ Pure Monte Carlo: trust the rollout.
        \item $\lambda = 0.95$ (typical): looks about $\frac{1}{1-\gamma\lambda} \approx 17$ steps ahead.
    \end{itemize}
    \end{column}
    \begin{column}{0.5\linewidth}
    \begin{tikzpicture}
    \begin{axis}[
        width=\linewidth, height=3.6cm,
        xmin=-1, xmax=40, ymin=0, ymax=1.05,
        xlabel={steps ahead $l$}, ylabel={weight on $\delta_{t+l}$},
        tick label style={font=\tiny}, label style={font=\scriptsize},
        legend columns=3, legend style={font=\tiny, draw=none, fill=none, at={(0.5,1.02)}, anchor=south}]
        \addplot[only marks, mark=*, mark size=1pt, red, domain=0:39, samples=40] {0.99^x};
        \addlegendentry{$\lambda = 1$}
        \addplot[only marks, mark=*, mark size=1pt, blue, domain=0:39, samples=40] {(0.99*0.95)^x};
        \addlegendentry{$\lambda = 0.95$}
        \addplot[only marks, mark=*, mark size=1pt, gray, domain=0:39, samples=40] {(0.99*0.7)^x};
        \addlegendentry{$\lambda = 0.7$}
    \end{axis}
    \end{tikzpicture}
    \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Computing GAE: walk backwards}
    A rollout is $M$ steps long, and may contain episode ends.
    Write the sum recursively, compute in reverse.
    \begin{block}{GAE recursion}
        \vspace{-0.4cm}
        \begin{align*}
            A_t &= \delta_t + \gamma\lambda\,(1-d_{t+1})\,A_{t+1}, \qquad A_{M-1} = \delta_{M-1} \\
            \delta_t &= r_{t+1} + \gamma\,(1-d_{t+1})\,\Vold(s_{t+1}) - \Vold(s_t)
        \end{align*}
    \end{block}
    \begin{center}
    \begin{tikzpicture}[x=11.5mm, cell/.style={draw, thick, minimum size=8.5mm, font=\normalsize}]
        \foreach \i in {0,...,3} { \node[cell, fill=red!10] (n\i) at (\i,0) {$A_{\i}$}; }
        \foreach \i in {4,...,7} { \node[cell, fill=blue!10] (n\i) at (\i,0) {$A_{\i}$}; }
        \node[cell, dashed, minimum width=15mm] (b) at (8.55,0) {$\Vold(s_8)$};
        \foreach \i/\j in {0/1, 1/2, 2/3, 4/5, 5/6, 6/7}
            { \draw[<-, thick] (n\i.north) to[bend left=50] node[above, font=\footnotesize] {$\gamma\lambda$} (n\j.north); }
        \draw[<-, thick] (n7.north) to[bend left=40] node[above, font=\footnotesize] {bootstrap (via $\delta_7$)} (b.north);
        \draw[red, very thick, dashed] (3.5,-0.6) -- (3.5,0.45);
        \draw[<-, thick] (n3.north) to[bend left=50] coordinate[pos=0.5] (cut) (n4.north);
        \draw[red, line width=2pt] ($(cut)+(-2.5mm,-2.5mm)$) -- ($(cut)+(2.5mm,2.5mm)$);
        \draw[red, line width=2pt] ($(cut)+(-2.5mm,2.5mm)$) -- ($(cut)+(2.5mm,-2.5mm)$);
        \node[red, font=\footnotesize] at (3.5,1.3) {episode ends ($d_4 = 1$)};
        \draw[->, thick, gray] (7.5,-0.75) -- (-0.5,-0.75) node[midway, below, font=\footnotesize] {compute right to left};
    \end{tikzpicture}
    \end{center}
    \pause
    \begin{itemize}
        \item $d_{t+1} = 1$: the episode ended, so don't bootstrap from $s_{t+1}$,
        don't leak information over the boundary.
        \item End of the rollout: $\Vold(s_M)$ stands in for everything we didn't see.
        \item Needs the steps \textbf{in time order}: do it once, right after the rollout.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{What does the critic learn from?}
    GAE also gives the critic a better target than $G_t$:
    \begin{block}{Critic target ($\lambda$-return)}
        $$
        G^\lambda_t = A_t + V_{\pth_\old}(s_t)
        $$
    \end{block}
    \pause
    \begin{itemize}
        \item Works on cut-off rollouts: no need to see the end of the episode.
        \item $\lambda = 0$: target is $r_{t+1} + \gamma V_{\pth_\old}(s_{t+1})$, i.e.\ TD(0).
    \end{itemize}
    \pause
    \begin{block}{Critic loss}
        $$
        L^{VF}(\pth) = \hat{\Ex}_t\Big[\big(G^\lambda_t - V_\pth(s_t)\big)^2 \Big]
        $$
    \end{block}
    $G^\lambda_t$ uses the critic \textit{at rollout time} ($\pth_\old$): 
    no gradient flows through it (like DQN's target network).
\end{frame}

\section{Clipping}

\begin{frame}
    \frametitle{Reusing a batch, and why it's dangerous}
    The actor's joy is now
    $
    J(\bth) = \hat{\Ex}_t\big[r_t(\bth)\,A_t\big]
    $
    with
    $
    r_t(\bth) = \frac{\pi_\bth(a_t|s_t)}{\pi_{\bth_\old}(a_t|s_t)}.
    $
    \begin{itemize}
        \pause
        \item First step: $\bth = \bth_\old$, so $r_t = 1$ and
        $\nabla_\bth r_t = \nabla_\bth \log \pi_\bth(a_t|s_t)$.
        This is exactly the VPG step.
        \pause
        \item We want to take \textit{many} steps on the same batch. It was expensive to collect.
        \pause
        \item But the batch describes trajectories following $\pi_{\bth_\old}$.
        Importance sampling only fixed the action probs, not the states (that $\pi_\bth$ maybe would never visit).
        The further $\pi_\bth$ moves, the more wrong $J(\bth)$ becomes.
        \pause
        \item $J(\bth)$ rewards large updates: if $A_t > 0$, it grows
        the more we push $r_t$ up.
    \end{itemize}
    \pause
    \vspace{0.2cm}
    We want: take the gain from moving a little, then stop.
\end{frame}

\begin{frame}
    \frametitle{Clipping: stop once you've moved enough}
    \begin{block}{PPO clipped joy}
        $$
        J^{CLIP}(\bth) = \hat{\Ex}_t\Big[\min\big(r_t(\bth)\,A_t,\;
        \text{clip}(r_t(\bth), 1-\eps, 1+\eps)\,A_t\big)\Big]
        $$
    \end{block}
    with $\eps \approx 0.2$. Plotting one term against $r_t$:
    \begin{columns}
    \begin{column}{0.48\linewidth}
    \centering
    \begin{tikzpicture}
    \begin{axis}[
        width=5.2cm, height=3.4cm, axis lines=middle,
        xmin=0, xmax=2, ymin=0, ymax=1.6,
        xtick={0.8,1.2}, xticklabels={$1\!-\!\eps$, $1\!+\!\eps$}, ytick=\empty,
        xlabel={$r_t$}, title={\scriptsize $A_t > 0$},
        tick label style={font=\tiny}, label style={font=\scriptsize}]
        \addplot[dashed, gray, domain=1.2:1.6] {x};
        \addplot[very thick, blue, domain=0:1.2] {x};
        \addplot[very thick, blue, domain=1.2:2] {1.2};
    \end{axis}
    \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\linewidth}
    \centering
    \begin{tikzpicture}
    \begin{axis}[
        width=5.2cm, height=3.4cm, axis lines=middle,
        xmin=0, xmax=2, ymin=-1.6, ymax=0,
        xtick={0.8,1.2}, xticklabels={$1\!-\!\eps$, $1\!+\!\eps$}, ytick=\empty,
        xlabel={$r_t$}, title={\scriptsize $A_t < 0$},
        tick label style={font=\tiny}, label style={font=\scriptsize}]
        \addplot[dashed, gray, domain=0.4:0.8] {-x};
        \addplot[very thick, red, domain=0:0.8] {-0.8};
        \addplot[very thick, red, domain=0.8:2] {-x};
    \end{axis}
    \end{tikzpicture}
    \end{column}
    \end{columns}
    \pause
    \begin{itemize}
        \item $A_t > 0$: gain for raising $\pi_\bth(a_t|s_t)$, but only up to
        $(1+\eps)\times$ its old value. Past that, the gradient is zero.
        \item $A_t < 0$: gain for lowering it, but only down to $(1-\eps)\times$.
        \pause
        \item The $\min$ only ever removes gain. If $\pi_\bth$ moved the \textit{wrong} way,
        the full gradient pulls it back.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Entropy bonus}
    $$
    H_{\bth}(s_t) = -\sum_{a \in \A} \pi_\bth(a|s_t) \log \pi_\bth(a|s_t)
    $$
    \begin{itemize}
        \item $0$ for a deterministic policy; $\log|\A|$ for a uniformly random one.
        \pause
        \item Add entropy $H_{\bth}(s_t)$ as bonus to objective to encourage exploration
        \pause
        \item No dependency on $a_t$ \emph{actually} taken.
        \item Debugging: if entropy crashes to 0 early, learning stalled.
    \end{itemize}
\end{frame}

\section{Putting it together}

\begin{frame}
    \frametitle{The PPO objective}
    \begin{block}{PPO joy (maximise)}
        $$
        J^{PPO}(\bth, \pth) = \hat{\Ex}_t\Big[
        \red{J^{CLIP}_t(\bth)}
        - c_{VF}\, \blue{\big(V_\pth(s_t) - G^\lambda_t\big)^2}
        + c_H\, \red{H_{\bth}(s_t)}
        \Big]
        $$
    \end{block}
    \begin{itemize}
        \item \red{Actor terms}: clipped policy gradient, plus the entropy bonus.
        \item \blue{Critic term}: regress $V_\pth(s_t)$ onto $G^\lambda_t$.
        \item Actor and critic share a backbone, so one optimiser step on $J^{PPO}$
        updates both heads, and both losses shape the shared features.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{What exactly do we learn from?}
    Each row of the batch is one step of one environment. Three kinds of quantity:
    \vspace{0.2cm}
    \begin{center}
    \small
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{l|l|l}
        \textbf{Stored during rollout} & \textbf{Computed after rollout} & \textbf{Recomputed per minibatch} \\ \hline
        $s_t,\ a_t,\ r_{t+1},\ d_{t+1}$ & $A_t$ (GAE) & $\pi_\bth(a_t|s_t)$, hence $r_t(\bth)$ \\
        $\log \pi_{\bth_\old}(a_t|s_t)$ & $G^\lambda_t = A_t + V_{\pth_\old}(s_t)$ & $H_\bth(s_t)$ \\
        $V_{\pth_\old}(s_t)$ & & $V_\pth(s_t)$ \\ \hline
        \multicolumn{2}{c|}{\textit{frozen: from the networks at rollout time}} & \textit{live: carries gradients}
    \end{tabular}
    \end{center}
    % \pause
    % \begin{itemize}
    %     \item Only the third column has gradients. The first two are just numbers.
    % \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{PPO Algorithm}
    \textbf{Input:} $N$ parallel envs, rollout length $M$, epochs $K$, learning rate $\alpha$

    \textbf{Algorithm: repeat:}
    \begin{enumerate}
        \item \textbf{Rollout:} run the current policy in all $N$ envs for $M$ steps.
        For each step store
        $$
        \big(s_t,\ a_t,\ r_{t+1},\ d_{t+1},\ \log \pi_{\bth_\old}(a_t|s_t),\ V_{\pth_\old}(s_t)\big)
        $$
        plus $V_{\pth_\old}(s_M)$ to bootstrap from.
        \item \textbf{GAE:} one backwards pass over each env's $M$ steps gives
        $A_t$ and $G^\lambda_t$.
        \item \textbf{Learn:} for $K$ epochs:
        \begin{itemize}
            \item shuffle all $NM$ steps into minibatches $\mathcal{B}$
            \item for each $\mathcal{B}$:
            $(\bth, \pth) \leftarrow (\bth, \pth) + \alpha \nabla J^{PPO}$, averaged over $\mathcal{B}$
        \end{itemize}
        \item \textbf{Discard} the batch. The envs are \textit{not} reset:
        the next rollout carries on mid-episode.
    \end{enumerate}
\end{frame}

    \frametitle{Where does the data go?}
    \begin{center}

\begin{frame}
    \frametitle{PPO in one slide}
    PPO is VPG with importance sampling, plus:
    \begin{enumerate}
        \item \textbf{Critic and advantage:} credit an action for being better than usual,
        not for being lucky to be in a good episode.
        \item \textbf{GAE:} advantages from short rollouts, by summing the critic's surprises.
        $\lambda$ trades noise against bias.
        \item \textbf{Clipping:} lets us reuse each batch for $K$ epochs without
        moving too far from the policy that collected it.
        \item \textbf{Entropy bonus:} keep exploring/prevent early policy collapse.
    \end{enumerate}
    \pause
    \vspace{0.3cm}
    Is it principled? Partly. Baselines are exact; per-step importance sampling is an
    approximation; clipping and the entropy bonus are heuristics. Works very well!
    \vspace{0.3cm}

    % \textbf{Looking ahead to RLHF:} the clipped ratio carries straight over to RL on LLMs.
    % GRPO even drops the critic and GAE: it estimates advantages by comparing several
    % answers to the same prompt.
\end{frame}
