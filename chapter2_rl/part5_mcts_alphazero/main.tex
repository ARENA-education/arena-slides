%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Planning with a simulator}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}
    \frametitle{A new setting: we know the rules}
    \begin{columns}
    \begin{column}{0.62\linewidth}
    So far (DQN, VPG, PPO) the agent had to learn the world from experience.
    \textbf{Model-free.}

    \vspace{0.3cm}
    For a board game we \textbf{have the simulator}. We can try a move, look at the result,
    and take it back, before playing for real. \textbf{Model-based: plan, then act.}

    \vspace{0.3cm}
    Today: \textbf{Connect 4}
    \begin{itemize}
        \item Two players, alternating turns, zero-sum
        \item Deterministic, perfect information
        \item Outcome $z \in \{+1, 0, -1\}$ (win / draw / loss), no discounting
        \item 7 columns, at most 42 plies (one \emph{ply} = one player's move)
    \end{itemize}
    \end{column}
    \begin{column}{0.36\linewidth}
    \centering
    \cfour[0.36]{3/0, 2/0, 3/2, 4/1}{3/1, 4/0, 2/1}

    \vspace{0.3cm}
    \small
    Connect 4 is \textbf{solved}: a perfect solver gives the optimal move from any
    position. A free ground truth to grade our agent against.
    \end{column}
    \end{columns}
\end{frame}


\begin{frame}
    \frametitle{Why not just search everything?}
    \begin{center}
        \includegraphics[width=0.55\linewidth]{img/minimax.png}
    \end{center}
    \textbf{Minimax:} expand every move to the end of the game, then back the values up.
    My best move is the one where the opponent's best reply is worst for me.

    \pause
    \vspace{0.2cm}
    \textbf{Problem:} the tree is huge. Connect 4 has about $4.5 \times 10^{12}$ positions;
    Go has more than atoms in the universe. We also can't write a good
    evaluation function for a mid-game position by hand.

    \pause
    \vspace{0.2cm}
    \textbf{Monte Carlo Tree Search:} don't search exhaustively. \textit{Sample}.
    Grow the tree \textbf{asymmetrically}, spending simulations on the moves that look promising.
\end{frame}


\begin{frame}
    \frametitle{Vanilla MCTS}
    \begin{center}
        \includegraphics[width=0.5\textwidth]{img/ch25-mcts.png}
    \end{center}
    \vspace{-0.2cm}
    Repeat many times:
    \begin{enumerate}
        \item \textbf{Select:} from the root, follow a \textit{tree policy} down to a leaf.
        \item \textbf{Expand:} add a new child node.
        \item \textbf{Simulate:} play \textit{random} moves to the end of the game.
        \item \textbf{Backup:} add the result to the visit counts $N$ and value sums $W$ on the path.
    \end{enumerate}
    Then play the \textbf{most-visited} root move.
\end{frame}


\begin{frame}
    \frametitle{The classic tree policy: UCB / UCT}
    Balance \textit{exploitation} (moves that did well) vs \textit{exploration}
    (moves we know little about):
    \begin{block}{UCB1 applied to trees (UCT)}
    $$
    \UCB(s, a) = \underbrace{Q(s, a)}_{\text{exploit}} +
    \underbrace{c \sqrt{\frac{\ln N(s)}{N(s, a)}}}_{\text{explore}}
    $$
    \end{block}
    \begin{itemize}
        \item $Q(s,a)$: mean outcome of $a$ so far; \; $N(s) = \sum_{a'} N(s, a')$
        \item The error of $Q(s,a)$ shrinks like $1/\sqrt{N(s,a)}$, so the bonus tracks the uncertainty
        \item $\ln N(s)$ grows slowly: every move is tried infinitely often, but exploitation
        eventually dominates. UCT converges to minimax as simulations $\to \infty$
    \end{itemize}
    \pause
    \textbf{What's missing?} Random rollouts are a noisy, weak value estimate; every move starts
    out equally plausible; and nothing is \textbf{learned} between searches.
\end{frame}

\begin{frame}
    \frametitle{One network, two heads}
    AlphaZero keeps the tree search and adds a network $f_\bth(s) = \big(p_\bth(\cdot|s),\, v_\bth(s)\big)$:

        \node[box, minimum width=18mm, right=of body, yshift=-8mm] (critic) {critic head};
        \node[right=of actor, align=left] {$p_\bth(\cdot|s)$: \textit{which column?}};
        \node[right=of critic, align=left] {$v_\bth(s) \in [-1,1]$: \textit{who's winning?}};
        \draw[->, thick] (s) -- (body);
        \draw[->, thick] (body.east) -- (actor.west);
        \draw[->, thick] (body.east) -- (critic.west);
        \draw[->, thick] (actor.east) -- ++(4mm,0);
        \draw[->, thick] (critic.east) -- ++(4mm,0);
    \end{tikzpicture}
    \end{center}
    \pause
    Two changes to MCTS:
    \begin{enumerate}
        \item \textbf{No random rollouts.} At a new leaf, ask the \blue{critic} $v_\bth(s)$
        how the game will end.
        \item \textbf{A prior in selection.} The \red{actor} $p_\bth(\cdot|s)$ says which
        moves are worth searching first.
    \end{enumerate}
    \pause
    \vspace{0.1cm}
    Far fewer simulations needed: the prior prunes the search, and one network call
    replaces a whole random playout.
\end{frame}

    \end{column}
    \begin{column}{0.52\linewidth}
    \textbf{Nodes} are positions (board + who moves).
    \textbf{Edges} are moves, and carry the statistics:
    \begin{itemize}
        \item $N(s,a)$: visit count
        \item $W(s,a)$: sum of backed-up values
        \item $Q(s,a) = W(s,a) / \max(1, N(s,a))$
        \item $P(s,a) = p_\bth(a|s)$: the prior
    \end{itemize}
    \vspace{0.2cm}
    $W$ and $Q$ are from the point of view of \textbf{$s$'s mover}: the player choosing
    between these edges.

    \vspace{0.2cm}
    \small Children are created lazily: one new node per simulation.
    \end{column}
    \end{columns}
\end{frame}


\begin{frame}
    \frametitle{Select: PUCT}
    At each node, take the legal move with the highest score:
    \begin{block}{PUCT (Predictor + Upper Confidence Trees)}
        $$
        \PUCT(s,a) = \underbrace{Q(s,a)}_{\text{exploit}}
        + \underbrace{c_{\text{puct}} \cdot \red{P(s,a)} \cdot
        \frac{\sqrt{1 + \sum_{a'} N(s,a')}}{1 + N(s,a)}}_{\text{explore}}
        $$
    \end{block}
    \begin{itemize}
        \pause
        \item Exploration is large for moves the \red{prior likes} but we have \textbf{rarely tried},
        and shrinks as $N(s,a)$ grows.
        \item So the search first follows the network's suggestions, then increasingly trusts
        the values $Q$ it has actually measured. It can overrule the prior.
        \pause
        \item Why $\sqrt{1 + \sum N}$? On a node's first visit every $N = 0$ and $Q = 0$.
        Without the $+1$ every move scores $0$; with it, the score is $\propto P(s,a)$,
        so we follow the prior straight away.
        \item Stop when the chosen move has \textbf{no child yet}, or we reach a \textbf{finished game}.
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{Expand and evaluate: no rollouts}
    \textbf{Expand:} play the chosen move in the simulator and add the new node.

    \vspace{0.3cm}
    \pause
    \textbf{Evaluate:} what is the leaf worth, \textit{to the leaf's mover}?
    \begin{center}
    \small
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{l|l}
        \textbf{the leaf is\ldots} & \textbf{its value} \\ \hline
        new, and the game is still going & $v_\bth(\text{leaf})$, from the \blue{critic} \\
        new, and the move just ended the game & $-r$ \\
        a finished game, reached again & the $-r$ stored when it was created
    \end{tabular}
    \end{center}
    \pause
    \begin{itemize}
        \item $r$ is the reward to the player who \textit{just moved}: $+1$ win, $0$ draw.
        (You can't lose on your own move.) The leaf's mover is the \textit{other} player, hence $-r$.
        \item A new non-terminal leaf also gets its prior $P(\text{leaf}, \cdot)$: the softmax of
        the \red{actor's} logits over the \textbf{legal} columns.
    \end{itemize}
\end{frame}

    Walk from the leaf to the root. At each step up, \textbf{flip the sign}, then update the edge:
    $N(s,a) \mathrel{+}= 1$, $\;W(s,a) \mathrel{+}= v$.

\begin{frame}
    \frametitle{MCTS Algorithm}
    \textbf{Input:} root position $s_0$, network $f_\bth$, number of simulations $\nsims$

    \textbf{Algorithm:}
    \begin{enumerate}
        \item \textbf{Root:} evaluate $s_0$ to get its prior $P(s_0, \cdot) = p_\bth(\cdot|s_0)$.
        \item \textbf{Repeat $\nsims$ times:}
        \begin{enumerate}
            \item \textbf{Select:} from the root, follow $\argmax_a \PUCT(s,a)$ until the
            chosen edge has no child, or the node is a finished game.
            \item \textbf{Expand:} play that edge in the simulator; add the new node.
            \item \textbf{Evaluate:} leaf value $v$: $v_\bth$ for a live position, $-r$ for a finished one.
            \item \textbf{Backup:} walk to the root, negating $v$ at each step;
            $N \mathrel{+}= 1$, $W \mathrel{+}= v$ on every edge.
        \end{enumerate}
        \item \textbf{Return} the root visit counts $N(s_0, \cdot)$.
    \end{enumerate}
    \vspace{0.2cm}
    Each simulation adds \textbf{at most one node} and makes \textbf{one network call}.
\end{frame}

\begin{frame}
    \frametitle{Search makes a better policy}
    The root visit counts define the \textbf{tree policy}:
    $$
    \pi(a|s) = \frac{N(s,a)^{1/\tau}}{\sum_{a'} N(s,a')^{1/\tau}}
    \qquad
    \begin{cases}
        \tau = 1: & \pi \propto N \;\text{(explore)} \\
        \tau \to 0: & \text{most-visited move (play to win)}
    \end{cases}
    $$
    \pause
    $\pi$ is \textbf{stronger} than the raw prior $p_\bth$: it is $p_\bth$ plus lookahead.
    \begin{itemize}
        \pause
        \item Give MCTS a \textit{dummy} network ($v = 0$, uniform prior) and a win-in-one position.
        With 32 simulations it finds the winning column: the terminal reward backs up the tree.
        The search alone does the tactics.
        \pause
        \item Take a trained network and search more at play time (0, 1, 2, \ldots, 64 sims):
        accuracy against the perfect solver climbs, with \textbf{no retraining}.
    \end{itemize}
    \pause
    \vspace{0.2cm}
    \textbf{Idea:} train $p_\bth$ to imitate $\pi$. A better $p_\bth$ gives a better search,
    which gives a better $\pi$, and so on.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Learning from self-play}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}
    \frametitle{Self-play: one ply}
    The network plays both sides. In every game, at every ply:
    \begin{enumerate}
        \item \textbf{Search:} run MCTS from the current position $s_t$ (with root noise, next slide).
        \item \textbf{Target:} $\pi_t = $ root visit counts, normalised. No temperature here.
        \item \textbf{Act:} sample $a_t$ from the visit counts at temperature $\tau$ and step the game.
        Finished games restart.
        \item \textbf{Record} $(s_t,\ \pi_t,\ d_t,\ r_t)$: done flag and reward to the mover.
    \end{enumerate}
    \pause
    \vspace{0.2cm}
    The \textbf{training target} and the \textbf{move played} only coincide at $\tau = 1$.
    We use $\tau = 1$ on every ply. (AlphaGo Zero used $\tau = 1$ for the first 30 moves, then
    $\tau \to 0$; a schedule is a bonus exercise.)
    \pause
    \vspace{0.2cm}

    \textbf{Exploration problem:} MCTS mostly searches moves the network already likes, and the network
    only learns about moves the search explores. A young network can lock onto a few openings.
\end{frame}


\begin{frame}
    \frametitle{Root exploration noise}
    \begin{block}{Dirichlet noise, at the root only}
        $$
        P(s_0, a) \leftarrow (1-\epsilon)\, p_\bth(a|s_0) + \epsilon\, \eta_a,
        \qquad \eta \sim \text{Dir}(\alpha)
        $$
    \end{block}
    \begin{itemize}
        \item $\text{Dir}(\alpha)$ is a distribution over probability vectors (the simplex),
        with the same $\alpha$ for every column:
        \begin{itemize}
            \item $\alpha < 1$: spiky. Occasionally gives one ignored move a big boost
            \item $\alpha = 1$: uniform over the simplex
            \item $\alpha > 1$: flat. A mild perturbation around $(\frac{1}{7}, \dots, \frac{1}{7})$
        \end{itemize}
        \pause
        \item Only at the root: it changes \textit{which lines get explored}, without corrupting
        the value estimates deeper in the tree.
        \item Off at evaluation and when checking batched vs single-game search.
    \end{itemize}
    \vspace{0.2cm}
    \small
    We use $\epsilon = 0.25$, $\alpha = 10/7 \approx 1.43$ (AlphaZero: $0.3$ for chess, $0.03$ for Go).
\end{frame}


\begin{frame}
    \frametitle{Value targets: walk backwards}
    The critic's target $z_t$ is how the game actually ended, \textbf{for the mover at ply $t$}.
    \begin{block}{Backwards scan (negamax)}
        $$
        z_t = \begin{cases}
            r_t & \text{if ply $t$ ended the game} \\
            -z_{t+1} & \text{otherwise}
        \end{cases}
        $$
    \end{block}
    \pause
    Like \texttt{compute\_returns} from VPG ($\gamma = 1$), plus a sign flip each ply.
    One row of a generation (red wins on ply 4, then a new game starts):
    \begin{center}

\begin{frame}
    \frametitle{The loss: distil the search into the network}
    Each recorded ply gives a training example $(s, \pi, z)$:
    \begin{block}{AlphaZero loss}
        $$
        L(\bth) = \Ex_{(s, \pi, z) \sim \mathcal{D}}\Big[
        \red{-\sum_a \pi(a|s) \log p_\bth(a|s)}
        \;+\; c_v \, \blue{\big(v_\bth(s) - z\big)^2}
        \Big]
        $$
    \end{block}
    \red{Actor:} cross-entropy, imitate the search. \quad
    \blue{Critic:} squared error, predict the outcome.
    Plus weight decay (handled by the optimiser).
    \pause
    \begin{columns}
    \begin{column}{0.5\linewidth}
    \centering
    \begin{tikzpicture}[b/.style={draw, thick, rounded corners, font=\small, align=center}]
        \node[b] (net) at (0,0) {network\\$p_\bth, v_\bth$};
        \node[b] (mcts) at (2.6,0) {search\\(MCTS)};
        \node[b] (pi) at (2.6,-1.6) {targets\\$\pi, z$};
        \node[b] (tr) at (0,-1.6) {train};
        \draw[->, thick] (net) -- node[above, font=\scriptsize] {guides} (mcts);
        \draw[->, thick] (mcts) -- node[right, font=\scriptsize] {self-play} (pi);
        \draw[->, thick] (pi) -- (tr);
        \draw[->, thick] (tr) -- (net);
    \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\linewidth}
    \small
    \textbf{Teacher:} network + search. \textbf{Student:} the network alone.
    The teacher is bootstrapped from the student, but the search makes it a little stronger,
    even from a random init.
    \end{column}
    \end{columns}
    \pause
    \vspace{0.2cm}
    No policy gradient, no advantages, no importance ratio. Just supervised learning
    on targets the search produced.
\end{frame}


\begin{frame}
    \frametitle{Policy loss: why cross-entropy?}
    We want $p_\bth$ to match the search policy $\pi$:
    $$
    D_{\text{KL}}(\pi \,\|\, p_\bth) = \sum_{a} \pi(a) \log \frac{\pi(a)}{p_\bth(a)}
    $$
    Differentiating:
    \begin{align*}
    \nabla_\bth D_{\text{KL}} &= \nabla_\bth \Big[ \underbrace{\sum_a \pi(a) \log \pi(a)}_{-H(\pi),\ \text{const.\ w.r.t.\ } \bth} - \sum_a \pi(a) \log p_\bth(a) \Big] \\[0.2cm]
    &= -\nabla_\bth \sum_a \pi(a) \log p_\bth(a)
    \end{align*}

    $\Rightarrow$ Minimising KL $\Leftrightarrow$ minimising \textbf{cross-entropy}
    $-\sum_a \pi(a|s) \log p_\bth(a|s)$.

    \vspace{0.2cm}
    \small
    In code: \texttt{-(pi * F.log\_softmax(logits, -1)).sum(-1).mean()}.
\end{frame}

\section{Making it fast}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}
    \frametitle{Making it fast: root parallelism}
    One network call per simulation on a batch of size 1 is far too slow. Instead, run $B$ games
    at once, each with its \textbf{own independent tree}.
    \begin{center}
    \begin{tikzpicture}[
        nd/.style={circle, draw, minimum size=4.5mm, inner sep=0pt},
        scale=0.8, transform shape]
        \foreach \x/\c/\l in {-4.5/red/1, -1.5/green!60!black/2, 1.5/blue/3, 4.5/orange/B} {
            \begin{scope}[xshift=\x cm]
                \node[nd, fill=\c!25, draw=\c] (r) at (0,0) {};
                \node[nd, fill=\c!10, draw=\c] (a) at (-0.6,-0.8) {};
                \node[nd, fill=\c!10, draw=\c] (b) at (0.6,-0.8) {};
                \node[nd, fill=\c!40, draw=\c, thick] (l) at (0.6,-1.6) {};
                \draw[\c, thick] (r) -- (a); \draw[\c, thick] (r) -- (b); \draw[\c, thick] (b) -- (l);
                \node[font=\scriptsize, \c] at (0,0.5) {game \l};
                \draw[->, \c, thick] (l) -- (0.6,-2.5);
            \end{scope}
        }
        \node[font=\large] at (3,-0.8) {$\cdots$};
        \node[draw, thick, rounded corners, fill=gray!15, minimum width=10.5cm, minimum height=7mm]
            at (0,-2.9) {\small one batched forward pass $f_\bth$ over all $B$ leaves};
    \end{tikzpicture}
    \end{center}
    \pause
    Per simulation: \textbf{select + expand} in every tree, \textbf{one} network call on the
    $B$ leaves, \textbf{backup} in every tree.
    \begin{itemize}
        \item Every game has a \textbf{different} root. The trees never merge; they only interact
        through the shared network.
        \item After the search, sample one move per game and step all $B$ environments.
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{A stepping stone: batch the net, keep Python trees}
    \texttt{SimulatedBatchedMCTS}: $B$ ordinary \texttt{Node} trees, section 2's \texttt{select}/\texttt{backup}
    in a Python \texttt{for} loop, but \textbf{one} network call per simulation.
    \begin{itemize}
        \item Shared \texttt{obs\_pool} \texttt{(B, MAXN, 3, 6, 7)}: nodes store an integer \texttt{slot};
        the batch is a single gather
        \item Fixed-size batch: terminal leaves are forwarded too (output discarded)
    \end{itemize}

    \vspace{0.2cm}
    \begin{center}
    \footnotesize
    \begin{tabular}{r|r|r|r}
        $B$ & Python trees + batched net & fully vectorised & speedup \\
        \hline
        64 & 716 & 4{,}751 & $6.6\times$ \\
        1{,}024 & 689 & 78{,}325 & $114\times$ \\
        4{,}096 & 691 & 189{,}903 & $275\times$ \\
        65{,}536 & $\sim$700 & \textbf{229{,}872} & $\mathbf{328\times}$ \\
    \end{tabular}

    \vspace{0.1cm}
    \scriptsize env-steps/s, sims = 32, RTX A4000
    \end{center}

    \small
    Python tree updates are the bottleneck ($\sim$700 steps/s regardless of $B$).
    Vectorising select / expand / backup as well gives $\sim 300\times$.
    We \textbf{give} you the fully vectorised \texttt{BatchedMCTS} for training; building it is the bonus.
\end{frame}


\begin{frame}
    \frametitle{Why not tree parallelism?}
    \begin{columns}
    \begin{column}{0.5\textwidth}
    \begin{center}
    \begin{tikzpicture}[
        every node/.style={circle, draw, minimum size=6mm, font=\scriptsize},
        scale=0.75, transform shape
    ]
        \node[fill=gray!20] (root) at (0,0) {$s_0$};
        \node[fill=red!30] (a) at (-2,-1.3) {$s_1$};
        \node (b) at (0,-1.3) {$s_2$};
        \node[fill=blue!30] (c) at (2,-1.3) {$s_3$};
        \node[fill=red!30] (a1) at (-2.6,-2.6) {$s_4$};
        \node (a2) at (-1.4,-2.6) {$s_5$};
        \node[fill=green!30] (b1) at (-0.4,-2.6) {$s_6$};
        \node (b2) at (0.4,-2.6) {$s_7$};
        \node (c1) at (1.4,-2.6) {$s_8$};
        \node[fill=blue!30] (c2) at (2.6,-2.6) {$s_9$};
        \draw[->] (root) -- (a);
        \draw[->] (root) -- (b);
        \draw[->] (root) -- (c);
        \draw[->, red, thick] (a) -- (a1);
        \draw[->] (a) -- (a2);
        \draw[->, green!50!black, thick] (b) -- (b1);
        \draw[->] (b) -- (b2);
        \draw[->] (c) -- (c1);
        \draw[->, blue, thick] (c) -- (c2);
        \node[draw=none, red] at (-2.6,-3.3) {W1};
        \node[draw=none, green!50!black] at (-0.4,-3.3) {W2};
        \node[draw=none, blue] at (2.6,-3.3) {W3};
    \end{tikzpicture}
    \end{center}
    \end{column}
    \begin{column}{0.5\textwidth}
        \small
        One shared tree, many workers descending different branches.

        \vspace{0.2cm}
        \green{\checkmark} No duplicated work: all statistics pooled

        \red{$\times$} \textbf{Node contention:} workers read/write the same $N, W$ (especially the root)
    \end{column}
    \end{columns}

    \vspace{0.2cm}
    \small
    \begin{itemize}
        \item \textbf{Mutexes:} lock a node before updating it. Contention at popular nodes eats the speedup
        \item \textbf{Virtual loss:} on the way down, temporarily count the edge as a visited loss so other
        workers spread out; undo it during backup
        \item \textbf{Transpositions:} hash positions and share nodes $\Rightarrow$ a DAG, not a tree
    \end{itemize}
    DeepMind's AlphaZero searched this way, with the tree in optimised C++ shuttling batches to
    accelerators. A nightmare to vectorise in PyTorch.
\end{frame}


\begin{frame}
    \frametitle{Bonus: trees as tensors}
    Store all $B$ trees as flat tensors indexed \texttt{[game, node, action]}:

    \vspace{0.1cm}
    \footnotesize
    \begin{columns}[T]
    \begin{column}{0.56\textwidth}
    \begin{itemize}
        \item \texttt{N, W, P\ \ (B, MAXN+1, 7)}: per edge
        \item \texttt{child\ \ (B, MAXN+1, 7)}: $-1$ = unexpanded
        \item \texttt{parent, parent\_act\ \ (B, MAXN+1)}
        \item \texttt{obs\_pool, tomove, terminal, term\_val}
        \item \texttt{nptr\ \ (B,)}: next free slot (root = 0)
    \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
    \begin{itemize}
        \item \texttt{MAXN = sims + 2}: each simulation adds at most one node
        \item Backup follows \texttt{parent} pointers home (fine: a strict tree)
        \item Every phase is \texttt{gather} / \texttt{scatter} / \texttt{where} / \texttt{argmax}
    \end{itemize}
    \end{column}
    \end{columns}

    \vspace{0.3cm}

    \textbf{Lockstep over games of different shapes:}
    \begin{itemize}
        \item Leaves sit at different depths: descend for a fixed \texttt{MAXD} steps with a \texttt{done} mask
        \item Games not expanding this step write to a throwaway \textbf{dustbin} slot (index \texttt{MAXN})
        \item \textbf{Sync-free:} no \texttt{.item()} in the hot loop, so the GPU never stalls.
        One deliberate exception: break out of selection early once all games found their leaf ($2.25\times$ faster)
    \end{itemize}

    \vspace{0.2cm}
    Test: with noise off, batched and single-game search give \textbf{identical} visit counts.
\end{frame}

\section{Putting it together}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}
    \frametitle{AlphaZero Algorithm}
    \textbf{Input:} $B$ parallel games, $\nsims$ simulations per move, buffer of the last $K$ generations

    \textbf{Algorithm: repeat for each generation:}
    \begin{enumerate}
        \item \textbf{Self-play} with the network frozen. For 42 plies, in all $B$ games:
        \begin{itemize}
            \item MCTS with root noise $\to$ $\pi_t$; sample $a_t$ at temperature $\tau$; step the game
            \item store $(s_t,\ \pi_t,\ d_t,\ r_t)$; finished games restart
        \end{itemize}
        \item \textbf{Targets:} one backwards pass per game row gives $z_t$.
        Drop plies from games that didn't finish.
        \item \textbf{Buffer:} add the $(s, \pi, z)$ examples; evict the oldest generation.
        \item \textbf{Learn:} one pass over the buffer in shuffled minibatches $\mathcal{B}$:
        $$
        \bth \leftarrow \bth - \alpha \nabla_\bth L(\bth), \text{ averaged over } \mathcal{B}
        $$
        \item \textbf{Evaluate} against the perfect solver.
    \end{enumerate}
\end{frame}


\begin{frame}
    \frametitle{The default recipe}
    \begin{center}
    \small
    \renewcommand{\arraystretch}{1.15}
    \begin{tabular}{l|l}
        games per generation $B$ & 4096 \\
        simulations per move $\nsims$ & 16 \quad (64 for the strongest run) \\
        generations & 12 \quad ($\sim$50 for the full recipe) \\
        buffer & last $K = 4$ generations \\
        $c_{\text{puct}}$ & 1.0 \\
        root noise & $\epsilon = 0.25$, $\alpha = 10/7$ \\
        optimiser & AdamW, minibatch 1024, grad-norm clip 1.0 \\
        learning rate & $5 \times 10^{-3} \to 2 \times 10^{-5}$ (cosine) \\
    \end{tabular}
    \end{center}
    \vspace{0.3cm}
    \begin{itemize}
        \item About 85\% solver accuracy in 4--5 minutes on one GPU.
        \item In the tuning sweep, every gain came from \textbf{fresher self-play data}
        (higher lr, smaller buffer). Capacity, loss weights and regularisation made little difference.
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{How good is it? Ask a perfect solver}
    Connect 4 is solved, so we can grade moves instead of playing matches.
    \begin{itemize}
        \item About 6700 positions, each labelled by Pascal Pons' solver with its set of
        optimal moves. Only \textit{decisive} positions are kept: the choice of move must matter.
        \pause
        \item One batched forward pass over the set gives:
        \begin{itemize}
            \item \textbf{acc:} how often $\argmax_a p_\bth(a|s)$ is an optimal move
            \item \textbf{ce:} $-\log \sum_{a \text{ optimal}} p_\bth(a|s)$, a smoother signal
        \end{itemize}
        \pause
        \item This grades the \textbf{raw policy}, with no search. Searching at play time makes it
        stronger still.
    \end{itemize}
    \pause
    \vspace{0.2cm}
    Then play against it yourself in the browser: \texttt{play\_web(model, env)}.
\end{frame}


\begin{frame}
    \frametitle{Bonus: search at play time, and more}
    \begin{itemize}
        \item \textbf{Accuracy vs search budget:} replay the solver set with $M \in \{0, 1, 2, \dots, 64\}$
        simulations per move ($M = 0$: raw policy). Accuracy climbs with $M$.
        \item \textbf{Elo vs $\log(\text{sims})$:} round-robin the \textit{same} network at different budgets
        and fit Elo ratings. Strength is roughly \textbf{linear in $\log(\text{search})$}: each doubling
        of thinking time buys a roughly constant Elo gain.
        \item \textbf{Mirror augmentation:} flip the board left-right, reverse $\pi$, keep $z$.
        Free extra data (AlphaGo Zero used all 8 symmetries of Go).
        \item Temperature schedule, subtree reuse between moves, bigger networks,
        comparison with PPO self-play.
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{AlphaZero in one slide}
    \begin{enumerate}
        \item \textbf{One network, two heads:} $p_\bth$ (which move?) and $v_\bth$ (who's winning?),
        always from the \textbf{mover's} point of view.
        \item \textbf{MCTS guided by the network:} PUCT uses the prior to decide what to search;
        the critic replaces random rollouts. Values flip sign once per ply (negamax).
        \item \textbf{Search improves the policy:} visit counts $\pi$ beat the raw prior $p_\bth$.
        \item \textbf{Self-play makes the data:} $(s, \pi, z)$ for every ply, $z$ from one
        backwards pass.
        \item \textbf{Distillation:} cross-entropy onto $\pi$, MSE onto $z$. Repeat.
        \item \textbf{Engineering:} root parallelism over thousands of independent games,
        with trees, environment and network all on the GPU.
    \end{enumerate}
    \pause
    \vspace{0.3cm}
    Compared with PPO: no policy gradient at all. The planner is the teacher,
    and RL happens through the loop: better net $\to$ better search $\to$ better targets.
\end{frame}
