\section{Planning with a simulator}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}
    \frametitle{Known environment, unknown strategy}
    \begin{columns}
    \begin{column}{0.66\linewidth}
    So far the agent had to learn the world from experience.
    \textbf{Model-free.}

    \vspace{0.15cm}
    For a board game we \textbf{have the simulator}. \\ 
    We can \textbf{plan}: try many moves in the sim
    before playing for real. \\
    \textbf{Model-based: plan, then act.}

    \vspace{0.15cm}
    Today: \textbf{Connect 4}
    \begin{itemize}
        \item Two player zero-sum game
        \item Deterministic, perfect information
        \item Reward $r_t = +1$ to the player whose move \textbf{wins}, else $0$. $\gamma = 1$.
    \end{itemize}
    \end{column}
    \begin{column}{0.32\linewidth}
    \centering
    \cfour[0.36]{3/0, 2/0, 3/2, 4/1}{3/1, 4/0, 2/1}

    \vspace{0.15cm}
    \small
    Connect 4 is \textbf{strongly solved}: we have an oracle for perfect
    play from any position.
    \end{column}
    \end{columns}
    \textbf{Return:} the outcome \textbf{for the mover at ply $t$}, and the target for the \blue{critic}.
    An \textbf{alternating} sum to the game's last ply $T$, where only $r_T \neq 0$:
    \vspace{-0.2cm}
    $$
    z_t = r_t - r_{t+1} + r_{t+2} + \ldots = (-1)^{T-t}\, r_T
    $$
\end{frame}


\begin{frame}
    \frametitle{Why not just search everything?}
    \begin{center}
        \includegraphics[width=0.55\linewidth]{img/minimax.png}
    \end{center}
    \textbf{Minimax:} expand every move to the end of the game, then back the values up.
    Good for me $\equiv$ bad for you

    \pause
    \vspace{0.2cm}
    \textbf{Problem:} the tree is huge, and evaluating partial game nodes
    requires a heuristic $\equiv$ domain knowledge

    \pause
    \vspace{0.2cm}
    \textbf{Monte Carlo Tree Search:} don't search exhaustively. \textit{Sample}.
    Grow the tree \textbf{asymmetrically}, spending simulations on the moves that look promising.
\end{frame}

    \begin{enumerate}
        \item \textbf{No random rollouts.} At a new leaf, ask the \blue{critic} $v_\pth(s)$

\begin{frame}
    \frametitle{Select: PUCT}
    At each node, take the legal move with the highest score:
    \begin{block}{PUCT (Predictor + Upper Confidence Trees)}
        $$
        \PUCT(s,a) = \underbrace{Q(s,a)}_{\text{exploit}}
        + \underbrace{c_{\text{puct}} \cdot \red{p_{\bth}(a|s)} \cdot
        \frac{\sqrt{1 + N(s)}}{1 + N(s,a)}}_{\text{explore}},
        \quad N(s) = \sum_{a'} N(s,a')
        $$
    \end{block}
    \begin{itemize}
        \pause
        \item Exploration is large for moves the \red{prior likes} but we have \textbf{rarely tried},
        and shrinks as $N(s,a)$ grows.
        \item Eventually model exploits moves that are good, and known to be good (visited often).
        \pause
        % \item Why $\sqrt{1 + N(s)}$? On a node's first visit every $N = 0$ and $Q = 0$.
        % Without the $+1$ every move scores $0$; with it, the score is $\propto p_{\bth}$,
        % so we follow the prior straight away.
        % \item Stop when the chosen move has \textbf{no child yet}, or we reach a \textbf{finished game}.
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{Why this exploration term?}
    The bonus $c_{\text{puct}} \, p_\bth(a|s) \, \frac{\sqrt{1+N(s)}}{1 + N(s,a)}$ measures how \textbf{under-visited} a move is \textbf{relative to its prior}.
    Suppose visits were shared out in proportion to the prior, $N(s,a) \approx p_\bth(a|s)\, N(s)$
    (and drop $c_{\text{puct}}$):
    $$
    p_\bth(a|s) \, \frac{\sqrt{1+N(s)}}{1 + N(s,a)}
    \overset{\text{large } N}{\approx} p_\bth(a|s) \, \frac{\sqrt{N(s)}}{N(s,a)}
    \approx p_\bth(a|s) \, \frac{\sqrt{N(s)}}{p_\bth(a|s)\, N(s)}
    = \frac{1}{\sqrt{N(s)}}
    $$
    \pause
    \begin{itemize}
        \item A move with \textbf{less} than its prior share of visits gets a \textbf{bigger} bonus;
        one with \textbf{more} gets a \textbf{smaller} bonus.
        \item $Q(s,a)$ averages $N(s,a)$ samples, so its standard error is $\approx 1/\sqrt{N(s,a)}$.
        If visits follow the prior, $N(s,a) \approx p_\bth(a|s)\, N(s)$, error $\propto 1/\sqrt{N(s)}$, so PUCT behaves like
        \mbox{``estimate $+\ c_{\text{puct}} \times$ standard error''}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{MCTS Algorithm}
    \textbf{Input:} root position $s_0$, network $(p_\bth, v_\pth)$, number of simulations $\nsims$

    \textbf{Algorithm:}
    \begin{enumerate}
        \item \textbf{Root:} evaluate $s_0$ and cache its prior $p_\bth(\cdot|s_0)$.
        \item \textbf{Repeat $\nsims$ times:}
        \begin{enumerate}
            \item \textbf{Select:} from the root, follow $\argmax_a \PUCT(s,a)$ until the
            chosen edge has no child, or the node is a finished game.
            \item \textbf{Expand:} play that edge in the simulator; add the new node.
            \item \textbf{Evaluate:} leaf value $v$: $v_\pth$ for a live position, $-r$ for a finished one.
            \item \textbf{Backup:} walk to the root, negating $v$ at each step;
            $N \mathrel{+}= 1$, $W \mathrel{+}= v$ on every edge.
        \end{enumerate}
        \item \textbf{Return} the root visit counts $N(s_0, \cdot)$.
    \end{enumerate}
    \vspace{0.2cm}
    Each simulation adds \textbf{exactly one node} and makes \textbf{one network call}.
\end{frame}


\begin{frame}
    \frametitle{Search makes a better policy}
    The root visit counts define the \textbf{tree policy}:
    $$
    \pi(a|s) = \frac{N(s,a)}{N(s)}
    \qquad
    \begin{cases}
        \text{self-play:} & \text{sample } a \sim \pi(\cdot \mid s) \;\text{(explore)} \\
        \text{deployment:} & \argmax_a N(s,a) \;\text{(play to win)}
    \end{cases}
    $$
    \pause
    $\pi$ is \textbf{stronger} than the raw prior $p_\bth$: it is $p_\bth$ plus lookahead.
    \begin{itemize}
        \pause
        % \item Give MCTS a \textit{dummy} network ($v = 0$, uniform prior) and a win-in-one position.
        % With 32 simulations it finds the winning column.
        % \pause
        \item Can take an already-trained network, and increase the number of sims to improve performance
        (test-time compute.)
    \end{itemize}
    \pause
    \vspace{0.2cm}
    \textbf{Idea:} Distill $\pi$ into $p_\bth$. A better $p_\bth$ gives a better search,
    which gives a better $\pi$, and so on.
\end{frame}

\begin{frame}
    \frametitle{Self-play: one ply}
    The network plays both sides. In every game, at every ply:
    \begin{enumerate}
        \item \textbf{Search:} run MCTS from the current position $s_t$.
        \item \textbf{Target:} $\pi_t = $ root visit counts, normalised.
        \item \textbf{Act:} sample $a_t \sim \pi_t$ and step the game. Finished games restart.
        \item \textbf{Record} $(s_t,\ \pi_t,\ d_t,\ r_t)$: done flag and reward to the mover.
    \end{enumerate}
    \pause
    \vspace{0.2cm}
    \textbf{Exploration:} MCTS mostly searches moves the network already likes, and the network
    only learns about moves the search explores.
    \pause
    \begin{block}{Inject Dirichlet noise at the root}
        $$
        p_\bth(a|s_0) \leftarrow (1-\epsilon)\, p_\bth(a|s_0) + \epsilon\, \eta_a,
        \qquad \eta \sim \text{Dir}(\alpha)
        $$
    \end{block}
    % It changes which lines get explored, without corrupting the value estimates deeper in the tree.
    % Smaller $\alpha$ gives spikier noise.
\end{frame}


\begin{frame}
    \frametitle{Value targets: walk backwards}
    The critic's target is the return $z_t$: how the game actually ended, \textbf{for the mover at ply $t$}.
    Compute the alternating sum backwards:
    \begin{block}{Backwards scan (negamax)}
        $$
        z_t = \begin{cases}
            r_t & \text{if ply $t$ ended the game} \\
            -z_{t+1} & \text{otherwise}
        \end{cases}
        $$
    \end{block}

\begin{frame}
    \frametitle{AlphaZero Algorithm}
    Repeat:
    \begin{enumerate}
        \item \textbf{Self-play.} Freeze the network and play $B$ games against itself. At every move:
        \begin{itemize}
            \item run MCTS from the current position to get $\pi_t$;
            \item play $a_t \sim \pi_t$, and remember $(s_t, \pi_t)$.
        \end{itemize}
        \vspace{0.1cm}
        \item \textbf{Label.} At game end, label each $s_t$ with outcome $z_t$ for the mover at ply $t$.
        \vspace{0.1cm}
        \item \textbf{Learn.} Train on the recent $(s, \pi, z)$ examples by minimising 
        $$
        L(\bth, \pth) = \Ex_{(s, \pi, z) \sim \mathcal{D}}\Big[
        \red{-\sum_a \pi(a|s) \log p_\bth(a|s)}
        \;+\; c_v \, \blue{\big(v_\pth(s) - z\big)^2}
        \Big]
        $$
    \end{enumerate}
    \vspace{0.3cm}
    A better network guides the search, leading to a stronger $\pi$, which we distill back to the network.
\end{frame}
